import { AbsoluteFill, spring, useCurrentFrame } from 'remotion';

export const KineticSplitCaption = ({ words, fps = 60 }) => {
  const frame = useCurrentFrame();
  const activeIdx = words.findIndex(w => frame >= w.start && frame < w.end);
  
  const bounce = spring({ frame: frame - (words[activeIdx]?.start || 0), fps, config: { damping: 12 } });

  return (
    <AbsoluteFill style={{ justifyContent: 'center', alignItems: 'center' }}>
      <div className="caption-line" style={{ transform: `scale(${1 + bounce * 0.08})` }}>
        {words.map((w, i) => (
          <span key={i} className={i === activeIdx ? "highlight" : "dim"}>{w.text} </span>
        ))}
      </div>
    </AbsoluteFill>
  );
};
